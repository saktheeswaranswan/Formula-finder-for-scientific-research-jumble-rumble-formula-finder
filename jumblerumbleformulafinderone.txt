// ==========================================
// SYMBOLIC MODEL FINDER + MEGA RANDOM SYSTEM
// ==========================================

import android.content.ClipData;
import android.content.ClipboardManager;
import android.content.Context;

int maxRows = 20;

float[] xs = new float[maxRows];
float[] ys = new float[maxRows];
boolean[] filled = new boolean[maxRows];

int selectedRow = 0;
boolean selectingX = true;

String inputBuffer = "";

// Tracking
String bestExprError = "None";
String bestExprCorr = "None";

float bestError = 1.0;
float bestCorr = -1;

int iterations = 0;
boolean searching = false;

// UI
String copiedMsg = "";
int copiedTimer = 0;

// =======================
// KEYPAD
// =======================
float keypadX = 450;
float keypadY = 200;
float keySize = 80;

boolean dragging = false;
float offsetX, offsetY;

String[] keys = {
  "7","8","9",
  "4","5","6",
  "1","2","3",
  "0",".","-",
  "DEL","OK"
};

// =======================
// SETUP
// =======================
void setup(){
  size(900,1500);
  textSize(26);
}

// =======================
// DRAW
// =======================
void draw(){

  background(245);

  drawTable();
  drawResult();
  drawButtons();
  drawKeypad();

  if(searching) runSearchStep();
}

// =======================
// TABLE
// =======================
void drawTable(){

  float startY = 40;

  for(int i=0;i<maxRows;i++){

    float y = startY + i*40;

    if(i==selectedRow) fill(200,220,255);
    else fill(255);

    rect(20,y-25,400,35);

    fill(0);

    String xText = filled[i]?nf(xs[i],1,2):"-";
    String yText = filled[i]?nf(ys[i],1,2):"-";

    text("X: "+xText,30,y);
    text("Y: "+yText,200,y);
  }

  fill(0);
  text("Editing: "+(selectingX?"X":"Y"),450,60);
  text("Input: "+inputBuffer,450,100);
}

// =======================
// BUTTONS
// =======================
void drawButtons(){

  fill(0,200,100);
  rect(20,820,200,60);
  fill(255);
  text("START",70,860);

  fill(200,0,0);
  rect(240,820,200,60);
  fill(255);
  text("CLEAR",290,860);

  fill(0,120,255);
  rect(460,820,250,60);
  fill(255);
  text("RANDOM DATA",500,860);
}

// =======================
// KEYPAD
// =======================
void drawKeypad(){

  pushMatrix();
  translate(keypadX,keypadY);

  for(int i=0;i<keys.length;i++){

    int col=i%3;
    int row=i/3;

    float x=col*keySize;
    float y=row*keySize;

    fill(220);
    rect(x,y,keySize,keySize);

    fill(0);
    textAlign(CENTER,CENTER);
    text(keys[i],x+keySize/2,y+keySize/2);
  }

  popMatrix();
}

// =======================
// RESULT
// =======================
void drawResult(){

  float y=900;

  fill(0);
  text("Best (Error):",20,y);
  fill(0,0,200);
  text(bestExprError,20,y+40);

  fill(0);
  text("Error: "+nf(bestError,1,5),20,y+80);

  text("Best (Corr):",20,y+140);
  fill(0,0,200);
  text(bestExprCorr,20,y+180);

  fill(0);
  text("R: "+nf(bestCorr,1,5),20,y+220);
  text("Iter: "+iterations,20,y+260);

  if(copiedTimer > 0){
    fill(0,150,0);
    text("Copied: "+copiedMsg,20,y+320);
    copiedTimer--;
  }
}

// =======================
// MOUSE
// =======================
void mousePressed(){

  float y=900;

  if(mouseX>20 && mouseX<800 && mouseY>y+20 && mouseY<y+60){
    copyToClipboard(bestExprError);
    return;
  }

  if(mouseX>20 && mouseX<800 && mouseY>y+160 && mouseY<y+200){
    copyToClipboard(bestExprCorr);
    return;
  }

  if(mouseX>20 && mouseX<220 && mouseY>820 && mouseY<880){
    searching=true;
  }

  if(mouseX>240 && mouseX<440 && mouseY>820 && mouseY<880){
    clearAll();
  }

  if(mouseX>460 && mouseX<710 && mouseY>820 && mouseY<880){
    generateRandomData();
  }

  // keypad click
  for(int i=0;i<keys.length;i++){
    int col=i%3;
    int row=i/3;

    float x=keypadX+col*keySize;
    float yk=keypadY+row*keySize;

    if(mouseX>x && mouseX<x+keySize &&
       mouseY>yk && mouseY<yk+keySize){

      handleKey(keys[i]);
      return;
    }
  }

  // drag
  if(mouseX>keypadX && mouseX<keypadX+3*keySize &&
     mouseY>keypadY && mouseY<keypadY+5*keySize){

    dragging=true;
    offsetX=mouseX-keypadX;
    offsetY=mouseY-keypadY;
  }

  int row=int((mouseY-40)/40);
  if(row>=0 && row<maxRows){
    selectedRow=row;
  }
}

void mouseDragged(){
  if(dragging){
    keypadX=mouseX-offsetX;
    keypadY=mouseY-offsetY;
  }
}

void mouseReleased(){
  dragging=false;
}

// =======================
// 🔥 MEGA RANDOM GENERATOR
// =======================
void generateRandomData(){

  clearAll();

  int mode = int(random(6));

  float scaleX = pow(10, random(-3,6));
  float scaleY = pow(10, random(-2,5));

  float chaos = random(0.1,0.9);
  float r = random(3.6,4.0);

  for(int i=0;i<maxRows;i++){

    float t = map(i,0,maxRows-1,-1,1);
    float x = t * scaleX;

    float y=0;

    if(mode==0){
      y = (0.2*x*x + 0.01*x*x*x) * scaleY;
    }
    else if(mode==1){
      y = (sin(x*0.01) + 0.5*sin(x*0.1) + 0.25*sin(x)) * scaleY;
    }
    else if(mode==2){
      y = exp(x*0.0001) * scaleY;
    }
    else if(mode==3){
      chaos = r * chaos * (1-chaos);
      y = chaos * scaleY;
    }
    else if(mode==4){
      float f=0;
      float amp=1;
      float freq=0.01;

      for(int k=0;k<6;k++){
        f += amp*sin(x*freq);
        amp *= 0.5;
        freq *= 2;
      }

      y = f * scaleY;
    }
    else{
      chaos = r * chaos * (1-chaos);
      float f = sin(x*0.05)+0.3*sin(x*0.5);
      y = (f+chaos)*scaleY;
    }

    y += random(-0.05,0.05)*scaleY;

    if(abs(y)>1e6) y = (y>0?1:-1)*1e6;

    xs[i]=x;
    ys[i]=y;
    filled[i]=true;
  }
}

// =======================
// COPY
// =======================
void copyToClipboard(String text){

  if(text.equals("None")) return;

  ClipboardManager clipboard =
    (ClipboardManager)getActivity().getSystemService(Context.CLIPBOARD_SERVICE);

  ClipData clip = ClipData.newPlainText("model", text);
  clipboard.setPrimaryClip(clip);

  copiedMsg = text;
  copiedTimer = 120;
}

// =======================
// HANDLE KEYS
// =======================
void handleKey(String k){

  if(k.equals("DEL")){
    if(inputBuffer.length()>0){
      inputBuffer=inputBuffer.substring(0,inputBuffer.length()-1);
    }
  }
  else if(k.equals("OK")){

    if(inputBuffer.length()>0){

      float val=float(inputBuffer);

      if(selectingX){
        xs[selectedRow]=val;
        selectingX=false;
      }else{
        ys[selectedRow]=val;
        filled[selectedRow]=true;
        selectingX=true;
        selectedRow=min(selectedRow+1,maxRows-1);
      }

      inputBuffer="";
    }
  }
  else{
    inputBuffer+=k;
  }
}

// =======================
// SEARCH
// =======================
void runSearchStep(){

  for(int i=0;i<300;i++){

    Expr e=randomExpr(4);

    float err=computeError(e);
    float corr=computeCorrelation(e);

    iterations++;

    if(err<bestError){
      bestError=err;
      bestExprError=e.toString();
    }

    if(corr>bestCorr && err<bestError*2){
      bestCorr=corr;
      bestExprCorr=e.toString();
    }
  }
}

// =======================
// EXPRESSION ENGINE
// =======================
class Expr{

  int type;
  float val;
  String op;
  Expr left,right;

  Expr(){type=0;}
  Expr(float v){type=1;val=v;}

  Expr(String op,Expr a,Expr b){
    type=2;
    this.op=op;
    left=a;
    right=b;
  }

  float eval(float x){

    if(type==0) return x;
    if(type==1) return val;

    float a=left.eval(x);
    float b=(right!=null)?right.eval(x):0;

    if(abs(a)>1e6 || abs(b)>1e6) return 1;

    float r=0;

    if(op.equals("+")) r=a+b;
    else if(op.equals("-")) r=a-b;
    else if(op.equals("*")) r=a*b;
    else if(op.equals("/")){
      if(abs(b)<1e-6) return 1;
      r=a/b;
    }
    else if(op.equals("sin")) r=sin(a);

    if(abs(r)>1e6) return 1;

    return r;
  }

  public String toString(){

    if(type==0) return "x";
    if(type==1) return nf(val,1,2);

    if(op.equals("sin")) return "sin("+left+")";

    return "("+left+op+right+")";
  }
}

// =======================
// RANDOM EXPRESSION
// =======================
Expr randomExpr(int d){

  if(d==0){
    if(random(1)<0.7) return new Expr();
    return new Expr(random(-2,2));
  }

  int c=int(random(5));

  if(c==0)return new Expr("+",randomExpr(d-1),randomExpr(d-1));
  if(c==1)return new Expr("-",randomExpr(d-1),randomExpr(d-1));
  if(c==2)return new Expr("*",randomExpr(d-1),randomExpr(d-1));
  if(c==3)return new Expr("/",randomExpr(d-1),randomExpr(d-1));

  return new Expr("sin",randomExpr(d-1),null);
}

// =======================
// ERROR
// =======================
float computeError(Expr e){

  float err=0;
  int n=0;

  for(int i=0;i<maxRows;i++){
    if(filled[i]){

      float p=e.eval(xs[i]);
      float y=ys[i];

      float diff=(p-y)/(abs(y)+1);
      err+=diff*diff;

      n++;
    }
  }

  if(n==0) return 1;

  return constrain(sqrt(err/n),0,1);
}

// =======================
// CORRELATION
// =======================
float computeCorrelation(Expr e){

  float sx=0,sy=0,sxy=0,sx2=0,sy2=0;
  int n=0;

  for(int i=0;i<maxRows;i++){
    if(filled[i]){

      float yt=ys[i];
      float yp=e.eval(xs[i]);

      sx+=yp; sy+=yt;
      sxy+=yp*yt;
      sx2+=yp*yp;
      sy2+=yt*yt;

      n++;
    }
  }

  if(n<2) return 0;

  float num=n*sxy-sx*sy;
  float den=sqrt((n*sx2-sx*sx)*(n*sy2-sy*sy));

  if(abs(den)<1e-6) return 0;

  return num/den;
}

// =======================
// CLEAR
// =======================
void clearAll(){

  for(int i=0;i<maxRows;i++) filled[i]=false;

  inputBuffer="";
  bestExprError="None";
  bestExprCorr="None";

  bestError=1;
  bestCorr=-1;

  iterations=0;
  searching=false;
}
